\begin{solusi}(By Valentio Iverson)
    Notice that by AM-GM
    \begin{align*} 
        ab + c = ab + c(a + b + c) = (a + c)(b + c) = ab + c(a + b) + c^2 \geq ab + 2c\sqrt{ab} + c^2 = (\sqrt{ab} + c)^2
    \end{align*}
    Hence
    \begin{align*}
        \sumcyc \sqrt{ab+c} \ge \sumcyc (\sqrt{ab} + c) \sumcyc \sqrt{ab} + 1
    \end{align*}
\end{solusi}
